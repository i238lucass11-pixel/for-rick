"""
verificar_entorno.py — Full environment check for the pipeline.

Run:  py verificar_entorno.py
Checks: Python version, every dependency (version + real import), known
cross-compatibilities (numpy 2 vs vectorbt, Python 3.14 vs numba), and that
the pipeline files are present and compile.
"""
import importlib
import os
import py_compile
import sys

sys.stdout.reconfigure(encoding='utf-8')

OK, WARN, FAIL = '✅', '⚠️ ', '❌'
problems = []

print('=' * 68)
print('ENVIRONMENT CHECK — Quant Pipeline')
print('=' * 68)

# ---------- Python ----------
v = sys.version_info
print(f"\nPython: {sys.version.split()[0]}  ({sys.executable})")
if (v.major, v.minor) in [(3, 10), (3, 11), (3, 12), (3, 13)]:
    print(f"  {OK} version supported by vectorbt/numba (CI matrix: 3.10-3.13)")
else:
    print(f"  {WARN} Python {v.major}.{v.minor} OUTSIDE the vectorbt CI matrix "
          f"(3.10-3.13).")
    print(f"     numba (vectorbt's engine) usually takes ~1 year to support")
    print(f"     each new Python. If everything imports below, it works —")
    print(f"     but on strange numba crashes, migrate to 3.12.")
    problems.append(f"Python {v.major}.{v.minor} outside supported matrix")

# ---------- dependencies ----------
DEPS = [
    # (import module, pip name, suggested minimum, used by)
    ('numpy',      'numpy',      '1.26',   'entire pipeline'),
    ('pandas',     'pandas',     '2.0',    'entire pipeline'),
    ('scipy',      'scipy',      '1.11',   'Portfolio_Balancer (shrinkage)'),
    ('sklearn',    'scikit-learn', '1.3',  'Portfolio_Balancer (Ledoit-Wolf)'),
    ('vectorbt',   'vectorbt',   '1.1.0',  'Quant_Miner'),
    ('optuna',     'optuna',     '4.0',    'Quant_Miner'),
    ('talib',      'TA-Lib',     '0.5',    'Quant_Miner'),
    ('yfinance',   'yfinance',   '0.2.50', 'Quant_Miner, Portfolio_Balancer'),
    ('pyarrow',    'pyarrow',    '15.0',   'parquet outputs'),
    ('tabulate',   'tabulate',   '0.9',    'notebook (to_markdown)'),
    ('matplotlib', 'matplotlib', '3.8',    'notebook (plots)'),
]

def _ver_tuple(s):
    out = []
    for p in str(s).split('.')[:3]:
        num = ''.join(c for c in p if c.isdigit())
        out.append(int(num) if num else 0)
    return tuple(out)

print('\nDependencies:')
versions = {}
for mod, pipname, vmin, used_by in DEPS:
    try:
        m = importlib.import_module(mod)
        ver = getattr(m, '__version__', '?')
        versions[mod] = ver
        status = OK
        if ver != '?' and _ver_tuple(ver) < _ver_tuple(vmin):
            status = WARN
            problems.append(f"{pipname} {ver} < {vmin} suggested")
        print(f"  {status} {pipname:12s} {ver:12s} (min suggested {vmin})  — {used_by}")
    except ImportError:
        print(f"  {FAIL} {pipname:12s} NOT INSTALLED — {used_by}")
        print(f"        py -m pip install {pipname}")
        problems.append(f"{pipname} not installed")
    except Exception as e:
        print(f"  {WARN} {pipname:12s} import failed: {e}")
        problems.append(f"{pipname}: {e}")

# ---------- known cross-compatibilities ----------
print('\nCross-compatibilities:')
try:
    np_ver = _ver_tuple(versions.get('numpy', '0'))
    vbt_ver = _ver_tuple(versions.get('vectorbt', '0'))
    if np_ver >= (2, 0) and vbt_ver < (1, 0):
        print(f"  {FAIL} numpy 2.x with vectorbt 0.2x: KNOWN INCOMPATIBILITY.")
        print(f"        Either downgrade numpy (<2) or upgrade vectorbt (>=1.1).")
        problems.append('numpy 2 + vectorbt 0.2x')
    else:
        print(f"  {OK} numpy {versions.get('numpy','?')} + "
              f"vectorbt {versions.get('vectorbt','?')}: valid combination")
except Exception:
    pass

# yfinance: fragility reminder
print(f"  {WARN} yfinance depends on Yahoo's unofficial API — if downloads")
print(f"     start failing en masse, the first thing to try is:")
print(f"        py -m pip install --upgrade yfinance")

# ---------- pipeline files ----------
print('\nPipeline files (presence + compilation):')
FILES = [
    'Quant_Miner.py', 'Strategy_Assembler.py', 'Portfolio_Balancer.py',
    'validation.py',
    'precios_cache.py', 'broker_config.py', 'forex_swap.py',
    'asset_group_config.py',
]
JSONS = ['sleeve_map.json', 'sector_map.json']
for f in FILES:
    if not os.path.exists(f):
        print(f"  {WARN} {f}: not in this folder")
        continue
    try:
        py_compile.compile(f, doraise=True)
        print(f"  {OK} {f}")
    except py_compile.PyCompileError as e:
        print(f"  {FAIL} {f}: DOES NOT COMPILE — {e.msg[:80]}")
        problems.append(f"{f} does not compile")
import json as _json
for f in JSONS:
    try:
        _json.load(open(f, encoding='utf-8'))
        print(f"  {OK} {f}")
    except FileNotFoundError:
        print(f"  {WARN} {f}: not in this folder")
    except Exception as e:
        print(f"  {FAIL} {f}: invalid JSON — {e}")
        problems.append(f"{f} invalid")

# ---------- summary ----------
print('\n' + '=' * 68)
if problems:
    print(f"{WARN} {len(problems)} item(s) to review:")
    for p in problems:
        print(f"   - {p}")
    print("\nTo see which packages have newer versions available:")
    print("   py -m pip list --outdated")
else:
    print(f"{OK} Environment complete and consistent. Ready to mine.")
print('=' * 68)
